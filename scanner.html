/** * HELLOFRESH BOX SCANNER — Backend API (Google Apps Script) * ---------------------------------------------------------- * Lives inside the route spreadsheet. The scanner web page is hosted * separately (e.g. GitHub Pages) and calls this script as an API. * This script reads routes, logs scans from all phones into one tab, * and posts a compiled end-of-night report to Slack. * * LiveStatus board: * A per-box status board rebuilt from the route file at start of night. * Column D holds '' | 'SCANNED' | 'MISSING'. It is a PROJECTION of ScanLog * (ScanLog stays the source of truth), so a phone that dies mid-route can be * replaced by any other phone, which reads back what's already SCANNED. * * "Start Fresh Night" button (sheet menu): * One click after pasting a new route export: rebuilds LiveStatus (resets the * status column to empty), wipes yesterday's ScanLog, and bumps a version * stamp so every phone throws away its cached routes and re-fetches. */ // ════════════════ CONFIG — EDIT THESE ════════════════ var CONFIG = { ROUTE_SHEET_NAME: 'Sheet1', // Tab name that holds the daily route export ROUTE_COL: 1, // Column A = routeId CODE_COL: 22, // Column V = box code (printed as barcode) STOP_NAME_COL: 19, // Column S = locationName (customer) — shown in missing report SCAN_LOG_SHEET: 'ScanLog', // Auto-created. Do not edit by hand. FOUND_SHEET: 'FoundBoxes', // Auto-created. Out-of-route / set-aside boxes + where they are. LIVE_SHEET: 'LiveStatus', // Auto-created/rebuilt. Live per-box status board. API_KEY: 'hf-scan-d06c1ea9f124', // Must match API_KEY in scanner.html SLACK_WEBHOOK_URL: 'PASTE_YOUR_SLACK_INCOMING_WEBHOOK_URL_HERE', REPORT_HOUR: 22, // 24h clock, script timezone. 22 = 10pm. Nightly auto-report. REPORT_TITLE: 'Nightly Box Scan Report' }; // ═════════════════════════════════════════════════════ // ---------- Data version (for phone cache invalidation) ---------- // Stored in Script Properties. Bumped by "Start Fresh Night". Phones compare // their cached version to this; a mismatch forces a fresh route download. function getDataVersion_() { var v = PropertiesService.getScriptProperties().getProperty('DATA_VERSION'); return v || '0'; } function bumpDataVersion_() { var v = String(Date.now()); // monotonic, unique per click PropertiesService.getScriptProperties().setProperty('DATA_VERSION', v); return v; } // ---------- API entry points ---------- // The scanner page sends POST requests with a JSON body: // { key: API_KEY, action: 'getRouteData' | 'logScan' | 'confirmRoute' // | 'getRouteProgress' | 'getVersion' // | 'logFound' | 'getFoundList' | 'resolveFound', ...params } function doPost(e) { var req; try { req = JSON.parse(e.postData.contents); } catch (err) { return json_({ ok: false, error: 'Bad request' }); } if (req.key !== CONFIG.API_KEY) return json_({ ok: false, error: 'Invalid key' }); try { switch (req.action) { case 'getVersion': return json_({ ok: true, data: { version: getDataVersion_() } }); case 'getRouteData': return json_({ ok: true, data: getRouteData() }); case 'logScan': logScan(req.scan || {}); return json_({ ok: true }); case 'getRouteProgress': return json_({ ok: true, data: getRouteProgress_(req.route) }); case 'confirmRoute': return json_({ ok: true, data: confirmRoute(req.route, req.scanner) }); case 'logFound': return json_({ ok: true, data: logFound_(req) }); case 'getFoundList': return json_({ ok: true, data: getFoundList_(req) }); case 'resolveFound': return json_({ ok: true, data: resolveFound_(req) }); default: return json_({ ok: false, error: 'Unknown action' }); } } catch (err) { return json_({ ok: false, error: String(err.message || err) }); } } // Simple health check if someone opens the URL in a browser function doGet() { return json_({ ok: true, service: 'box-scanner-api' }); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ---------- Route data ---------- /** * Returns { version, routes: [{id, count}], codeToRoute: {CODE: routeId}, codeToStop: {CODE: name} } * Codes are uppercased + trimmed. Rows with no code (e.g. DEPOT rows) are skipped. * The version lets the phone detect a "fresh night" bump on the same call it * uses to load routes, not just via a separate getVersion poll. */ function getRouteData() { var sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.ROUTE_SHEET_NAME); if (!sheet) throw new Error('Route tab "' + CONFIG.ROUTE_SHEET_NAME + '" not found. Check CONFIG.ROUTE_SHEET_NAME.'); var values = sheet.getDataRange().getValues(); var codeToRoute = {}; var codeToStop = {}; var routeCounts = {}; for (var i = 1; i < values.length; i++) { // skip header var code = String(values[i][CONFIG.CODE_COL - 1] || '').trim().toUpperCase(); if (!code) continue; var route = String(values[i][CONFIG.ROUTE_COL - 1]).trim(); if (route === '') continue; codeToRoute[code] = route; codeToStop[code] = String(values[i][CONFIG.STOP_NAME_COL - 1] || '').trim(); routeCounts[route] = (routeCounts[route] || 0) + 1; } var routes = Object.keys(routeCounts) .sort(function (a, b) { return (Number(a) || 0) - (Number(b) || 0) || a.localeCompare(b); }) .map(function (r) { return { id: r, count: routeCounts[r] }; }); return { version: getDataVersion_(), routes: routes, codeToRoute: codeToRoute, codeToStop: codeToStop }; } // ---------- LiveStatus board (per-box status projection) ---------- /** * Finds (or creates) the LiveStatus tab. Header only if freshly created. * Layout: A=Route B=Code C=Stop D=Status ('' | SCANNED | MISSING) * E=Scanner (who scanned it) F=UpdatedAt */ function liveSheet_() { var ss = SpreadsheetApp.getActive(); var sheet = ss.getSheetByName(CONFIG.LIVE_SHEET); if (!sheet) { sheet = ss.insertSheet(CONFIG.LIVE_SHEET); sheet.appendRow(['Route', 'Code', 'Stop', 'Status', 'Scanner', 'UpdatedAt']); sheet.setFrozenRows(1); } return sheet; } /** * Rebuilds the LiveStatus board from the current route file, wiping all * statuses back to empty. This regenerates the status column from the new * sheet. Called by "Start Fresh Night". This is the only thing that clears status. */ function rebuildLiveStatus_() { var data = getRouteData(); var sheet = liveSheet_(); // Clear everything below the header. if (sheet.getLastRow() > 1) sheet.deleteRows(2, sheet.getLastRow() - 1); // One row per box, ordered by route then code so the board reads naturally. var rows = []; Object.keys(data.codeToRoute) .sort(function (a, b) { var ra = data.codeToRoute[a], rb = data.codeToRoute[b]; var na = Number(ra) || 0, nb = Number(rb) || 0; if (na !== nb) return na - nb; if (ra !== rb) return ra.localeCompare(rb); return a.localeCompare(b); }) .forEach(function (code) { rows.push([data.codeToRoute[code], code, data.codeToStop[code] || '', '', '', '']); }); if (rows.length) sheet.getRange(2, 1, rows.length, 6).setValues(rows); return rows.length; } /** * Reads the LiveStatus board into { map: {ROUTE|CODE: idx}, values: [...] }. */ function liveIndex_(sheet) { var out = { map: {}, values: [] }; if (sheet.getLastRow() < 2) return out; var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues(); out.values = values; for (var i = 0; i < values.length; i++) { var key = String(values[i][0]).trim() + '|' + String(values[i][1]).trim().toUpperCase(); out.map[key] = i; // 0-based index into values (row = i + 2) } return out; } /** * Stamp a single box as SCANNED on the board. Called from logScan for OK scans. * Writes against the box's route (col A) so "moved to correct lane" lands on the * correct route row. No-op if the board hasn't been built or the row is absent. */ function markLiveScanned_(route, code, scanner) { route = String(route).trim(); code = String(code).trim().toUpperCase(); if (!route || !code) return; var sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.LIVE_SHEET); if (!sheet || sheet.getLastRow() < 2) return; // board not built — resume falls back to ScanLog var idx = liveIndex_(sheet); var i = idx.map[route + '|' + code]; if (i === undefined) return; // this code isn't on this route's board if (String(idx.values[i][3]).toUpperCase() === 'SCANNED') return; // already done var row = i + 2; sheet.getRange(row, 4, 1, 3).setValues([['SCANNED', String(scanner || ''), new Date()]]); } /** * At route confirmation, stamp every still-empty row for that route as MISSING. * Leaves SCANNED rows untouched. Returns count marked missing. */ function markLiveMissingForRoute_(route) { route = String(route).trim(); var sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.LIVE_SHEET); if (!sheet || sheet.getLastRow() < 2) return 0; var idx = liveIndex_(sheet); var values = idx.values; var changed = 0; for (var i = 0; i < values.length; i++) { if (String(values[i][0]).trim() !== route) continue; var status = String(values[i][3]).toUpperCase(); if (status === '') { values[i][3] = 'MISSING'; values[i][5] = new Date(); changed++; } } if (changed) sheet.getRange(2, 1, values.length, 6).setValues(values); return changed; } /** * action: 'getRouteProgress' — returns codes already SCANNED for this route so * ANY phone can resume, even a fresh one replacing a dead phone. * Prefers the LiveStatus board; falls back to ScanLog if the board isn't built. */ function getRouteProgress_(route) { route = String(route || '').trim(); if (!route) return { route: route, scanned: [] }; var sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.LIVE_SHEET); if (sheet && sheet.getLastRow() >= 2) { var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues(); var scanned = []; values.forEach(function (r) { if (String(r[0]).trim() === route && String(r[3]).toUpperCase() === 'SCANNED') { scanned.push(String(r[1]).trim().toUpperCase()); } }); return { route: route, scanned: scanned, source: 'live' }; } // Fallback: reconstruct from today's OK scans in ScanLog. var ok = getTodaysOkScans_()[route] || {}; return { route: route, scanned: Object.keys(ok), source: 'scanlog' }; } // ---------- Scan logging ---------- function getLogSheet_() { var ss = SpreadsheetApp.getActive(); var sheet = ss.getSheetByName(CONFIG.SCAN_LOG_SHEET); if (!sheet) { sheet = ss.insertSheet(CONFIG.SCAN_LOG_SHEET); sheet.appendRow(['Timestamp', 'Scanner', 'Selected Route', 'Code', 'Status', 'Expected Route']); sheet.setFrozenRows(1); } return sheet; } /** * Called from the phone for every scan. * status: OK | WRONG_ROUTE | NOT_FOUND | DUPLICATE * Side effect: an OK scan also stamps the LiveStatus board (auto-updates it). */ function logScan(scan) { var lock = LockService.getScriptLock(); lock.waitLock(10000); // many phones writing at once try { var route = String(scan.route || ''); var code = String(scan.code || '').trim().toUpperCase(); var status = String(scan.status || ''); getLogSheet_().appendRow([ new Date(), String(scan.scanner || 'Unknown'), route, code, status, String(scan.expectedRoute || '') ]); // Keep the live board current the moment a box is confirmed present. if (status === 'OK') markLiveScanned_(route, code, scan.scanner); } finally { lock.releaseLock(); } return true; } /** * Called when a driver taps "Confirm route complete". * Looks at TODAY's OK scans across ALL phones and returns boxes still missing. * Also stamps MISSING onto the LiveStatus board for any unscanned box. * * Note: the scanner's "I moved it to the correct lane" action logs an OK scan * against the box's CORRECT route, so those boxes count as present here. */ function confirmRoute(route, scanner) { route = String(route).trim(); var data = getRouteData(); var expected = {}; Object.keys(data.codeToRoute).forEach(function (code) { if (data.codeToRoute[code] === route) expected[code] = true; }); var scanned = getTodaysOkScans_()[route] || {}; var missing = Object.keys(expected) .filter(function (code) { return !scanned[code]; }) .map(function (code) { return { code: code, stop: data.codeToStop[code] || '' }; }); // Reflect the outcome on the live board (empty rows -> MISSING). markLiveMissingForRoute_(route); logScan({ scanner: scanner, route: route, code: '— ROUTE CONFIRMED —', status: missing.length ? 'CONFIRMED_WITH_MISSING(' + missing.length + ')' : 'CONFIRMED_COMPLETE', expectedRoute: route }); return { route: route, total: Object.keys(expected).length, scannedCount: Object.keys(scanned).length, missing: missing }; } /** {route: {code: true}} for today's OK scans, all phones combined */ function getTodaysOkScans_() { var sheet = getLogSheet_(); var out = {}; if (sheet.getLastRow() < 2) return out; var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues(); var tz = Session.getScriptTimeZone(); var today = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd'); values.forEach(function (row) { if (!(row[0] instanceof Date)) return; if (Utilities.formatDate(row[0], tz, 'yyyy-MM-dd') !== today) return; if (row[4] !== 'OK') return; var route = String(row[2]); out[route] = out[route] || {}; out[route][String(row[3])] = true; }); return out; } // ---------- Found boxes (out-of-route / set-aside) ---------- /** Finds (or creates) the FoundBoxes tab. */ function foundSheet_() { var ss = SpreadsheetApp.getActive(); var sheet = ss.getSheetByName(CONFIG.FOUND_SHEET); if (!sheet) { sheet = ss.insertSheet(CONFIG.FOUND_SHEET); sheet.appendRow(['Timestamp', 'Code', 'BelongsToRoute', 'Location', 'Scanner', 'ScannedUnderRoute', 'Resolved', 'ResolvedBy', 'ResolvedTime']); sheet.setFrozenRows(1); } return sheet; } /** * action: 'logFound' — save an out-of-route / unknown box and where it was left, * so a runner can collect it later. Params: code, belongsToRoute, location, * scanner, scannedUnderRoute. */ function logFound_(req) { var code = String(req.code || '').trim().toUpperCase(); if (!code) throw new Error('Missing code'); var lock = LockService.getScriptLock(); lock.waitLock(10000); try { foundSheet_().appendRow([ new Date(), code, String(req.belongsToRoute || 'UNKNOWN'), String(req.location || ''), String(req.scanner || ''), String(req.scannedUnderRoute || ''), false, '', '' ]); } finally { lock.releaseLock(); } return { code: code, saved: true }; } /** * action: 'getFoundList' — return found boxes. Unresolved only by default; * pass includeResolved: true to get everything. Newest first. */ function getFoundList_(req) { var sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.FOUND_SHEET); if (!sheet || sheet.getLastRow() < 2) return { found: [] }; var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 9).getValues(); var includeResolved = req && req.includeResolved; var out = []; for (var i = 0; i < rows.length; i++) { var r = rows[i]; var resolved = (r[6] === true) || (String(r[6]).toUpperCase() === 'TRUE'); if (includeResolved || !resolved) { out.push({ code: String(r[1]), belongsToRoute: String(r[2]), location: String(r[3]), scanner: String(r[4]), scannedUnderRoute: String(r[5]), resolved: resolved, time: r[0] instanceof Date ? r[0].getTime() : 0 }); } } out.sort(function (a, b) { return b.time - a.time; }); // newest first return { found: out }; } /** * action: 'resolveFound' — mark all open rows for a code as collected. * Params: code, scanner. */ function resolveFound_(req) { var code = String((req && req.code) || '').trim().toUpperCase(); if (!code) return { code: code, resolved: 0 }; var sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.FOUND_SHEET); if (!sheet || sheet.getLastRow() < 2) return { code: code, resolved: 0 }; var lock = LockService.getScriptLock(); lock.waitLock(10000); var n = 0; try { var rng = sheet.getRange(2, 1, sheet.getLastRow() - 1, 9); var rows = rng.getValues(); for (var i = 0; i < rows.length; i++) { var resolved = (rows[i][6] === true) || (String(rows[i][6]).toUpperCase() === 'TRUE'); if (String(rows[i][1]).toUpperCase() === code && !resolved) { rows[i][6] = true; rows[i][7] = String((req && req.scanner) || ''); rows[i][8] = new Date(); n++; } } if (n) rng.setValues(rows); } finally { lock.releaseLock(); } return { code: code, resolved: n }; } // ---------- Nightly Slack report ---------- function sendNightlyReport() { var data = getRouteData(); var okScans = getTodaysOkScans_(); var wrongAndUnknown = getTodaysProblemScans_(); var tz = Session.getScriptTimeZone(); var today = Utilities.formatDate(new Date(), tz, 'EEE, MMM d yyyy'); // Expected codes per route var expectedByRoute = {}; Object.keys(data.codeToRoute).forEach(function (code) { var r = data.codeToRoute[code]; expectedByRoute[r] = expectedByRoute[r] || []; expectedByRoute[r].push(code); }); var lines = ['*' + CONFIG.REPORT_TITLE + ' — ' + today + '*', '']; var totalExpected = 0, totalScanned = 0, totalMissing = 0; var routesWithIssues = 0; Object.keys(expectedByRoute) .sort(function (a, b) { return (Number(a) || 0) - (Number(b) || 0) || a.localeCompare(b); }) .forEach(function (route) { var expected = expectedByRoute[route]; var scanned = okScans[route] || {}; var missing = expected.filter(function (c) { return !scanned[c]; }); totalExpected += expected.length; totalScanned += Object.keys(scanned).length; totalMissing += missing.length; if (missing.length === 0) { lines.push(':white_check_mark: *Route ' + route + '* — ' + expected.length + '/' + expected.length + ' boxes scanned'); } else { routesWithIssues++; lines.push(':x: *Route ' + route + '* — ' + Object.keys(scanned).length + '/' + expected.length + ' scanned, *' + missing.length + ' missing:*'); missing.forEach(function (code) { var stop = data.codeToStop[code]; lines.push(' • `' + code + '`' + (stop ? ' — ' + stop : '')); }); } }); // Wrong-route / unknown scans if (wrongAndUnknown.length) { lines.push(''); lines.push(':warning: *Wrong-route / unknown scans today:*'); wrongAndUnknown.forEach(function (s) { if (s.status === 'WRONG_ROUTE') { lines.push(' • `' + s.code + '` scanned into route ' + s.route + ' but belongs to route ' + s.expectedRoute + ' (' + s.scanner + ')'); } else { lines.push(' • `' + s.code + '` scanned into route ' + s.route + ' — *not found in today\'s file* (' + s.scanner + ')'); } }); } // Found boxes still set aside on the wrong lane (unresolved) var found = getFoundList_({}).found; if (found.length) { lines.push(''); lines.push(':package: *Boxes set aside on wrong lane (not yet collected):*'); found.forEach(function (f) { lines.push(' • `' + f.code + '` belongs to route ' + f.belongsToRoute + ' — at ' + (f.location || 'location not noted') + (f.scannedUnderRoute ? ' (seen on route ' + f.scannedUnderRoute + ')' : '') + ' — ' + (f.scanner || 'unknown')); }); } lines.push(''); lines.push('*Summary:* ' + totalScanned + '/' + totalExpected + ' boxes scanned · ' + totalMissing + ' missing · ' + routesWithIssues + ' route(s) with issues' + (found.length ? ' · ' + found.length + ' set aside' : '')); postToSlack_(lines.join('\n')); } /** Today's WRONG_ROUTE and NOT_FOUND scans (deduped) */ function getTodaysProblemScans_() { var sheet = getLogSheet_(); var out = [], seen = {}; if (sheet.getLastRow() < 2) return out; var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues(); var tz = Session.getScriptTimeZone(); var today = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd'); values.forEach(function (row) { if (!(row[0] instanceof Date)) return; if (Utilities.formatDate(row[0], tz, 'yyyy-MM-dd') !== today) return; var status = String(row[4]); if (status !== 'WRONG_ROUTE' && status !== 'NOT_FOUND') return; var key = status + '|' + row[3] + '|' + row[2]; if (seen[key]) return; seen[key] = true; out.push({ scanner: String(row[1]), route: String(row[2]), code: String(row[3]), status: status, expectedRoute: String(row[5]) }); }); return out; } function postToSlack_(text) { if (!CONFIG.SLACK_WEBHOOK_URL || CONFIG.SLACK_WEBHOOK_URL.indexOf('http') !== 0) { throw new Error('Slack webhook URL not set in CONFIG.SLACK_WEBHOOK_URL'); } UrlFetchApp.fetch(CONFIG.SLACK_WEBHOOK_URL, { method: 'post', contentType: 'application/json', payload: JSON.stringify({ text: text }) }); } // ---------- Start Fresh Night (one button does it all) ---------- /** * The button you click AFTER pasting the new route export into the sheet. * In order: wipe yesterday's ScanLog -> rebuild the LiveStatus board (resets the * status column to empty from the new sheet) -> bump the version so every phone * drops its cached routes and re-downloads. Confirms first (destructive). */ function startFreshNight() { var ui = SpreadsheetApp.getUi(); var resp = ui.alert( 'Start a fresh night?', 'This will:\n' + ' 1. Clear ALL of yesterday\'s scans from "' + CONFIG.SCAN_LOG_SHEET + '"\n' + ' 2. Rebuild the "' + CONFIG.LIVE_SHEET + '" board from the current sheet (all statuses reset to empty)\n' + ' 3. Force every phone to reload the new routes\n\n' + 'Do this only after you\'ve pasted in tonight\'s route export. Continue?', ui.ButtonSet.YES_NO); if (resp !== ui.Button.YES) return; // 1. Wipe yesterday's scan log. var log = getLogSheet_(); if (log.getLastRow() > 1) log.deleteRows(2, log.getLastRow() - 1); // 2. Rebuild the live board from the (new) route file. var n = rebuildLiveStatus_(); // 3. Invalidate every phone's cached routes. var v = bumpDataVersion_(); ui.alert('Fresh night ready ✅', n + ' boxes loaded onto the ' + CONFIG.LIVE_SHEET + ' board.\n' + 'Yesterday\'s scans cleared.\n' + 'Phones will pull the new routes on their next load (version ' + v + ').', ui.ButtonSet.OK); } // ---------- Setup helpers (spreadsheet menu) ---------- function onOpen() { SpreadsheetApp.getUi().createMenu('📦 Box Scanner') .addItem('🚀 Start Fresh Night (after pasting new routes)', 'startFreshNight') .addSeparator() .addItem('Send Slack report now', 'sendNightlyReport') .addItem('Install nightly auto-report trigger', 'installNightlyTrigger') .addToUi(); } function installNightlyTrigger() { // Remove existing copies first ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'sendNightlyReport') ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger('sendNightlyReport') .timeBased().atHour(CONFIG.REPORT_HOUR).everyDays(1).create(); SpreadsheetApp.getUi().alert('Done! The Slack report will send automatically every day around ' + CONFIG.REPORT_HOUR + ':00 (' + Session.getScriptTimeZone() + ').'); }
